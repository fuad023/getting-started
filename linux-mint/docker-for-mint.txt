# more help @ https://oneuptime.com/blog/post/2026-02-08-how-to-install-docker-on-linux-mint
# official help @ https://docs.docker.com/engine/install/ubuntu/

# to add docker's gpp key
sudo install -m 0755 -d /etc/apt/keyrings
sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
sudo chmod a+r /etc/apt/keyrings/docker.asc

# to add docker repo
UBUNTU_CODENAME=$(cat /etc/upstream-release/lsb-release | grep CODENAME | cut -d '=' -f 2)
echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.asc] https://download.docker.com/linux/ubuntu $UBUNTU_CODENAME stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# to install docker engine
sudo apt update
sudo apt install docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

# to start and enable docker
sudo systemctl status docker
sudo systemctl start docker
sudo systemctl enable docker

# test run
sudo docker run hello-world

# to run docker w/o sudo
sudo usermod -aG docker $USER
docker ps

# to get docker desktop [optional]
wget https://desktop.docker.com/linux/main/amd64/docker-desktop-amd64.deb?utm_source=docker&utm_medium=webreferral&utm_campaign=docs-driven-download-linux-amd64
sudo apt install ./docker-desktop-amd64.deb
