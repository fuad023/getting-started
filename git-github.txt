################################ create ssh connection; follow either #1 or #2

#1 the noob approach

ssh-keygen -t ed25519 -C [your-email]
# press enter to set default location
# passphrase: set-some-password
eval "$(ssh-agent -s)"
ssh-add ~/.ssh/id_ed25519
cat ~/.ssh/id_ed25519.pub
# put publuc key @ GitHub > Settings > SSH and GPG keys > New SSH Key, with some title

#2 the pro approach B)

sudo apt install gh

gh auth login
# account to log into: github.com
# protocol: ssh
# generate ssh key: Y
# passphrase: set-some-password
# title: set-some-title
# authenticate: login w/ browser
