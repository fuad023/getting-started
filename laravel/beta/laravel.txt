################################ install composer; follow either #1 or #2

#1 only if you a pro
# to install composer using php dependency manager
curl -sS https://getcomposer.org/installer | sudo php -- --install-dir=/usr/local/bin --filename=composer
composer --version

#2 the hard way
# to download the installer script
php -r "copy('https://getcomposer.org/installer', '/tmp/composer-setup.php');"

# to verify the installer
HASH=`curl -sS https://composer.github.io/installer.sig`
php -r "if (hash_file('SHA384', '/tmp/composer-setup.php') === '$HASH') { echo 'Installer verified'; } else { echo 'Installer corrupt'; unlink('/tmp/composer-setup.php'); } echo PHP_EOL;"

# PROCEED ONLY WITH VERIFIED INSTALLER

# to install composer globally
sudo php /tmp/composer-setup.php --install-dir=/usr/local/bin --filename=composer

# to remove the installer
php -r "unlink('/tmp/composer-setup.php');"

composer -v # verify

################################ install laravel
